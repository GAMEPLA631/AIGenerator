Kód na pripojenie do iných webov

Tento kód si môže ktokoľvek vložiť do svojho webu na GitHube:

<script src="https://cdn.jsdelivr.net/npm/gun/gun.js"></script>
<script>
  const db = Gun(['https://gun-manhattan.herokuapp.com/gun', 'https://peer.wallie.io/gun']).get('claund_fast_db_v1');

  // Uloženie
  function ulozData(kluc, text) {
      db.get(kluc).put(text);
  }

  // Čítanie
  db.map().on((data, kluc) => {
      console.log("Nové dáta v databáze:", kluc, data);
  });
</script>
